import os
from pathlib import Path
from dataclasses import dataclass, field
from typing import Dict, Any

@dataclass
class Paths:
    """File and directory paths configuration."""
    data_folder: Path = field(default_factory=lambda: Path(os.getenv('EDIT_DATA_FOLDER', './data')))
    editroom_folder: Path = field(default_factory=lambda: Path(os.getenv('EDITROOM_DATA_FOLDER', './editroom')))
    blender_path: Path = field(default_factory=lambda: Path(os.getenv('BLENDER_PATH', '/usr/bin/blender')))

    def __post_init__(self):
        """Validate paths exist."""
        for path_name, path_value in self.__dict__.items():
            if not path_value.exists():
                raise FileNotFoundError(f"Required path {path_name} not found: {path_value}")

@dataclass
class ModelConfig:
    """Model configuration parameters."""
    max_objects: Dict[str, int] = field(default_factory=lambda: {
        'bedroom': 12,
        'livingroom': 21,
        'diningroom': 21
    })

    normalization_bounds: Dict[str, Any] = field(default_factory=dict)
